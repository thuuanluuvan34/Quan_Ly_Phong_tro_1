package com.example.quanlyphongtro

import android.app.Activity
import android.content.Intent
import android.os.Bundle
import android.widget.Button
import android.widget.TextView
import android.widget.Toast

class OwnerActivity : Activity() {

    private lateinit var owner: String

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        setContentView(R.layout.activity_owner)

        // Lấy tài khoản đã đăng nhập
        val username = intent.getStringExtra("username")

        if (username.isNullOrBlank()) {
            Toast.makeText(
                this,
                "Không tìm thấy thông tin đăng nhập",
                Toast.LENGTH_SHORT
            ).show()

            finish()
            return
        }

        owner = username

        // Hiển thị lời chào
        findViewById<TextView>(R.id.txtOwnerWelcome).text =
            "Xin chào, $owner!"

        // Hàm mở các màn hình quản lý
        fun openScreen(target: Class<out Activity>) {
            val screenIntent = Intent(this, target)
            screenIntent.putExtra("owner", owner)
            startActivity(screenIntent)
        }

        // Quản lý phòng
        findViewById<Button>(R.id.btnRooms).setOnClickListener {
            openScreen(RoomsActivity::class.java)
        }

        // Quản lý người thuê
        findViewById<Button>(R.id.btnTenants).setOnClickListener {
            openScreen(TenantsActivity::class.java)
        }

        // Quản lý điện
        findViewById<Button>(R.id.btnElectric).setOnClickListener {
            openScreen(ElectricActivity::class.java)
        }

        // Hóa đơn và thanh toán
        findViewById<Button>(R.id.btnBills).setOnClickListener {
            openScreen(BillsActivity::class.java)
        }

        // Thống kê doanh thu
        findViewById<Button>(R.id.btnStatistics).setOnClickListener {
            openScreen(StatsActivity::class.java)
        }

        // Đăng xuất
        findViewById<Button>(R.id.btnOwnerLogout).setOnClickListener {
            val loginIntent = Intent(
                this,
                LoginActivity::class.java
            )

            loginIntent.flags =
                Intent.FLAG_ACTIVITY_NEW_TASK or
                        Intent.FLAG_ACTIVITY_CLEAR_TASK

            startActivity(loginIntent)
            finish()
        }
    }

    override fun onResume() {
        super.onResume()

        if (!::owner.isInitialized) {
            return
        }

        try {
            RentalDb(this).use { db ->

                val rooms = db.rooms(owner)
                val bills = db.bills(owner)

                // Tổng số phòng
                findViewById<TextView>(R.id.txtRoomCount).text =
                    "Tổng số phòng: ${rooms.size}"

                // Hóa đơn chưa thanh toán đủ
                val unpaidCount = bills.count {
                    it.paid < it.total
                }

                findViewById<TextView>(R.id.txtDebtCount).text =
                    "Hóa đơn chưa thanh toán: $unpaidCount"

                // Doanh thu điện đã thu
                val totalRevenue = bills.sumOf {
                    it.paid
                }

                findViewById<TextView>(R.id.txtRevenue).text =
                    "Tiền điện đã thu: $totalRevenue VNĐ"
            }
        } catch (e: Exception) {
            Toast.makeText(
                this,
                "Không thể tải dữ liệu Dashboard",
                Toast.LENGTH_LONG
            ).show()
        }
    }
}
